func Pow

Pow(NaN, y) = NaN
Pow(x, NaN) = NaN
Pow(±0, y) = ±Inf，当 y 为小于 0 的奇数时
Pow(±0, -Inf) = +Inf
Pow(±0, +Inf) = +0
Pow(±0, y) = +Inf，当 y 为小于 0 的有限数且不是奇数时
Pow(±0, y) = ±0，当 y 为大于 0 的奇数时
Pow(±0, y) = +0，当 y 为大于 0 的有限数且不是奇数时
Pow(-1, ±Inf) = 1
Pow(x, +Inf) = +Inf，当 |x| > 1 时
