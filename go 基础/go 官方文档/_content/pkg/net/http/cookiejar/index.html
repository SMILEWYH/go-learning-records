▾ 示例

代码：

// 启动一个服务器来提供 cookie。
ts := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
    if cookie, err := r.Cookie("Flavor"); err != nil {
        http.SetCookie(w, &http.Cookie{Name: "Flavor", Value: "Chocolate Chip"})
    } else {
        cookie.Value = "Oatmeal Raisin"
        http.SetCookie(w, cookie)
    }
}))
defer ts.Close()

u, err := url.Parse(ts.URL)
if err != nil {
    log.Fatal(err)
}

// 所有 cookiejar 使用者都应导入 "golang.org/x/net/publicsuffix"
jar, err := cookiejar.New(&cookiejar.Options{PublicSuffixList: publicsuffix.List})
if err != nil {
    log.Fatal(err)
}

client := &http.Client{
    Jar: jar,
}

if _, err = client.Get(u.String()); err != nil {
    log.Fatal(err)
}

fmt.Println("After 1st request:")
for _, cookie := range jar.Cookies(u) {
    fmt.Printf("  %s: %s\n", cookie.Name, cookie.Value)
}

if _, err = client.Get(u.String()); err != nil {
    log.Fatal(err)
}

fmt.Println("After 2nd request:")
for _, cookie := range jar.Cookies(u) {
    fmt.Printf("  %s: %s\n", cookie.Name, cookie.Value)
}

输出：

After 1st request:
  Flavor: Chocolate Chip
After 2nd request:
  Flavor: Oatmeal Raisin

type Options 1.1

Options 是创建新 Jar 时使用的选项。 

type Options struct {
    // PublicSuffixList 是一个公共后缀列表，用于判断
    // HTTP 服务器能否为某个域名设置 cookie。
    //
    // nil 值是有效的，并可能适用于测试，但它
    // 不安全：这意味着 foo.co.uk 的 HTTP 服务器可以
    // 为 bar.co.uk 设置 cookie。
    PublicSuffixList PublicSuffixList
}

type PublicSuffixList 1.1

PublicSuffixList 提供域名的公共后缀。例如：

  • “example.com”的公共后缀是“com”，
  • “foo1.foo2.foo3.co.uk”的公共后缀是“co.uk”，以及
  • “bar.pvt.k12.ma.us”的公共后缀是“pvt.k12.ma.us”。

PublicSuffixList 的实现必须能够供多个 goroutine（轻量级协程）安全地并发使用。

始终返回空字符串的实现是有效的，并可能适用于测试，但它不安全：这意味着 foo.com 的 HTTP 服务器可以为 bar.com 设置 cookie。

公共后缀列表的一种实现在 golang.org/x/net/publicsuffix 包中。 

type PublicSuffixList interface {
    // PublicSuffix 返回 domain 的公共后缀。
    //
    // TODO：明确调用方和被调用方中哪一方负责处理 IP
    // 地址、开头和结尾的点、大小写敏感性，
    // 以及 IDN/Punycode。
    PublicSuffix(domain string) string

    // String 返回此公共后缀列表来源的描述。
    // 描述通常包含时间戳
    // 或版本号之类的信息。
    String() string
}
