tls 包部分实现了 RFC 5246 规定的 TLS 1.2 和 RFC 8446 规定的 TLS 1.3。

FIPS 140-3 模式

程序处于 FIPS 140-3 模式时，本包的行为相当于只实现了 SP 800-140C 和 SP 800-140D 批准的协议版本、密码套件、签名算法、证书公钥类型和大小，以及密钥交换和派生算法。其他选项会被静默忽略而不参与协商，或被拒绝。该集合可能取决于 GOFIPS140 选择的 FIPS 140-3 Go 密码模块所支持的算法，并可能随 Go 版本变化。

type Certificate

Certificate 是由一个或多个证书组成的证书链，叶子证书在前。 

type Certificate struct {
    Certificate [][]byte
    // PrivateKey 包含与以下位置的公钥对应的私钥：
    // Leaf。它必须实现 [crypto.Signer]，其 PublicKey 类型为 RSA、ECDSA、Ed25519
    // （TLS 1.2+）或 ML-DSA（TLS 1.3）。
    //
    // 对于 TLS 1.2 及更早版本的服务器，也可以实现 crypto.Decrypter，
    // 并使用 RSA PublicKey。
    //
    // 如果它实现了 [crypto.MessageSigner]，则在 TLS 1.2 及更高版本中
    // 使用 SignMessage，而不是 Sign。
    PrivateKey crypto.PrivateKey
    // SupportedSignatureAlgorithms 是可选列表，用于限制
    // PrivateKey 可以使用的签名算法。
    SupportedSignatureAlgorithms []SignatureScheme // Go 1.14
    // OCSPStaple 包含可选的 OCSP 响应，
    // 会提供给请求它的客户端。
    OCSPStaple []byte
    // SignedCertificateTimestamps 包含可选的签名证书时间戳列表，
    // 会提供给请求它的客户端。
    SignedCertificateTimestamps [][]byte // Go 1.5
    // Leaf 是叶子证书的解析形式，可以通过 x509.ParseCertificate
    // 初始化，以减少每次握手的处理量。如果为 nil，
    // 则在需要时解析叶子证书。
    Leaf *x509.Certificate
}

type ClientHelloInfo
				
				1.4
			

			
ClientHelloInfo 包含 ClientHello 消息中的信息，用于指导 GetCertificate 和 GetConfigForClient 回调中的应用逻辑。

			
type ClientHelloInfo struct {
    // CipherSuites 列出客户端支持的密码套件，例如
    // TLS_AES_128_GCM_SHA256、TLS_ECDHE_ECDSA_WITH_AES_128_GCM_SHA256。
    CipherSuites []uint16

    // ServerName 表示客户端请求的服务器名称，
    // 用于支持虚拟主机。只有客户端使用 SNI 时才会设置
    // ServerName（参见 RFC 4366 第 3.1 节）。
    ServerName string

    // SupportedCurves 列出客户端支持的密钥交换机制。
    // 在 TLS 1.3 中它改称为“支持的群”，参见 RFC 8446
    // 第 4.2.7 节及 [CurveID]。
    //
    // 在 TLS 1.2 及更早版本中，如果未使用支持的椭圆曲线扩展，
    // SupportedCurves 可能为 nil（参见 RFC 4492 第 5.1.1 节）。
    SupportedCurves []CurveID

    // SupportedPoints 列出客户端支持的点格式。
    // 只有使用支持的点格式扩展时才会设置 SupportedPoints
    // （参见 RFC 4492 第 5.1.2 节）。
    SupportedPoints []uint8

    // SignatureSchemes 列出客户端愿意验证的签名和
    // 哈希方案。只有使用签名算法扩展时才会设置 SignatureSchemes
    // （参见 RFC 5246 第 7.4.1.4.1 节）。
    SignatureSchemes []SignatureScheme // Go 1.8

    // SupportedProtos 列出客户端支持的应用协议。
    // 只有使用应用层协议协商扩展时才会设置 SupportedProtos
    // （参见 RFC 7301 第 3.1 节）。
    //
    // 服务器可以通过在 GetConfigForClient 的返回值中设置
    // Config.NextProtos 来选择协议。
    SupportedProtos []string // Go 1.8

    // SupportedVersions 列出客户端支持的 TLS 版本。
    // 对于低于 1.3 的 TLS 版本，该列表由客户端声明的
    // 最高版本推算而来，因此使用除最高版本之外的值
    // 时可能被拒绝。
    SupportedVersions []uint16 // Go 1.8

    // Extensions 列出客户端在 ClientHello 中
    // 提供的扩展 ID。
    Extensions []uint16 // Go 1.24

    // Conn 是该连接底层的 net.Conn。不得读取
    // 或写入此连接，否则会导致 TLS
    // 连接失败。
    Conn net.Conn // Go 1.8

    // HelloRetryRequest 表示此 ClientHello 是否作为
    // 对 HelloRetryRequest 消息的响应而发送。
    HelloRetryRequest bool // Go 1.26
    // 包含已过滤或未导出的字段
}

ClientSessionCache 是 ClientSessionState 对象的缓存，客户端可借此恢复与给定服务器的 TLS 会话。ClientSessionCache 的实现应能处理不同 goroutine 的并发调用。对于 TLS 1.2 及更早版本，只支持基于票据的恢复，不支持基于 SessionID 的恢复。在 TLS 1.3 中，两者合并为 PSK 模式，并通过此接口提供支持。

			
type ClientSessionCache interface {
    // Get 查找与给定键关联的 ClientSessionState。
    // 返回时，ok 为 true 表示找到。
    Get(sessionKey string) (session *ClientSessionState, ok bool)

    // Put 使用给定键将 ClientSessionState 添加到缓存。如果 TLS 1.3
    // 服务器提供多个会话票据，一次连接中可能调用多次。
    // 如果传入 nil *ClientSessionState，
    // 则应移除相应缓存条目。
    Put(sessionKey string, cs *ClientSessionState)
}

type Config
				
				
			

			
Config 结构用于配置 TLS 客户端或服务器。将它传给 TLS 函数后，不得再修改。Config 可以复用；tls 包也不会修改它。

			
type Config struct {
    // Rand 为连接提供熵源。
    // 如果 Rand 为 nil，TLS 使用 crypto/rand 包中的密码学随机读取器。
    // 该 Reader 必须能由多个 goroutine 安全使用。
    //
    // 已弃用：生产环境中应保留为 nil。并非所有 TLS
    // 配置都保证使用 Rand。测试代码可以改用
    // [testing/cryptotest.SetGlobalRandom]。
    Rand io.Reader

    // Time 返回当前时间，以自纪元以来的秒数表示。
    // 如果 Time 为 nil，TLS 使用 time.Now。
    Time func() time.Time

    // Certificates 包含要提供给连接对端的一个或多个证书链。
    // 会自动选择第一个符合对端要求的
    // 证书。
    //
    // 服务器配置必须设置 Certificates、GetCertificate 或
    // GetConfigForClient 中的一项。执行客户端身份认证的客户端可以设置
    // Certificates 或 GetClientCertificate。
    //
    // 注意：如果存在多个 Certificate，且它们都未设置
    // 可选的 Leaf 字段，选择证书会在每次握手中产生
    // 显著的性能开销。
    Certificates []Certificate

    // NameToCertificate 将证书名称映射到 Certificates
    // 中的一个元素。注意，证书名称可以采用
    // '*.example.com' 的形式，因此不一定是通常意义上的域名。
    //
    // 已弃用：NameToCertificate 只能为给定名称关联单个
    // 证书。保留此字段为 nil，让库从 Certificates 中
    // 选择第一条兼容的证书链。
    NameToCertificate map[string]*Certificate

    // GetCertificate 根据给定的 ClientHelloInfo
    // 返回 Certificate。只有客户端提供了 SNI
    // 信息，或 Certificates 为空时，才会调用它。
    //
    // 如果 GetCertificate 为 nil 或返回 nil，则从
    // NameToCertificate 获取证书。如果 NameToCertificate 为 nil，
    // 则使用 Certificates 中最合适的元素。
    //
    // Certificate 一经返回，就不应再修改。
    GetCertificate func(*ClientHelloInfo) (*Certificate, error) // Go 1.4

    // GetClientCertificate 非 nil 时，会在服务器向客户端
    // 请求证书时被调用。如果设置了此字段，就会
    // 忽略 Certificates 的内容。
    //
    // 如果 GetClientCertificate 返回错误，则中止握手，
    // 并返回该错误。否则，
    // GetClientCertificate 必须返回非 nil 的 Certificate。如果
    // Certificate.Certificate 为空，就不会向服务器
    // 发送证书。如果服务器不接受这种情况，可能会中止
    // 握手。
    //
    // 如果发生重新协商，或使用 TLS 1.3，同一连接中
    // 可能会多次调用 GetClientCertificate。
    //
    // Certificate 一经返回，就不应再修改。
    GetClientCertificate func(*CertificateRequestInfo) (*Certificate, error) // Go 1.8

    // GetConfigForClient 非 nil 时，会在收到客户端的
    // ClientHello 后被调用。它可以返回非 nil 的 Config，
    // 以更改处理此连接所用的 Config。如果
    // 返回的 Config 为 nil，就使用原 Config。
    // 此回调返回的 Config 此后不得修改。
    //
    // 如果 GetConfigForClient 为 nil，则对所有连接使用
    // 传给 Server() 的 Config。
    //
    // 如果返回的 Config 显式设置了 SessionTicketKey，或对
    // 返回的 Config 调用了 SetSessionTicketKeys，就使用这些密钥。
    // 否则，使用原 Config 的密钥（如果密钥自动管理，
    // 也可能进行轮换）。警告：这允许恢复最初通过父 Config
    // 或同级 Config 建立的连接，
    // 从而可能绕过返回的 Config 中设置的 [Config.VerifyPeerCertificate]
    // 回调。
    GetConfigForClient func(*ClientHelloInfo) (*Config, error) // Go 1.8

    // VerifyPeerCertificate 非 nil 时，会在 TLS 客户端或服务器
    // 完成正常证书验证之后被调用。
    // 它接收对端提供的原始 ASN.1 证书，
    // 以及正常处理过程中找到的已验证证书链。如果它返回
    // 非 nil 错误，则中止握手并返回该错误。
    //
    // 如果正常验证失败，会在考虑此回调之前中止
    // 握手。如果正常验证被禁用（客户端设置了 InsecureSkipVerify，
    // 或服务器的 ClientAuth 为 RequestClientCert 或 RequireAnyClientCert），
    // 则仍会考虑调用此回调，
    // 但 verifiedChains 参数始终为 nil。当
    // ClientAuth 为 NoClientCert 时，服务器不会调用此回调。
    // 如果 ClientAuth 为 RequestClientCert 或 VerifyClientCertIfGiven，
    // 服务器上的 rawCerts 可能为空。
    //
    // 恢复连接时不会调用此回调。警告：这包括在 [Config.Clone] 或
    // [Config.GetConfigForClient] 返回的 Config 与其父 Config 之间
    // 恢复的连接。如果不希望如此，
    // 请改用 [Config.VerifyConnection]，或设置 [Config.SessionTicketsDisabled]。
    //
    // 不应修改 verifiedChains 及其中的内容。
    VerifyPeerCertificate func(rawCerts [][]byte, verifiedChains [][]*x509.Certificate) error // Go 1.8

    // VerifyConnection 非 nil 时，会在 TLS 客户端或服务器完成
    // 正常证书验证及 VerifyPeerCertificate 调用之后
    // 被调用。如果它返回非 nil 错误，则中止握手，
    // 并返回该错误。
    //
    // 如果正常验证失败，会在考虑此回调之前中止
    // 握手。此回调对所有连接都会运行，
    // 包括恢复的连接，不受 InsecureSkipVerify 或 ClientAuth
    // 设置影响。
    VerifyConnection func(ConnectionState) error // Go 1.15

    // RootCAs 定义客户端验证服务器证书时
    // 使用的根证书颁发机构集合。
    // 如果 RootCAs 为 nil，TLS 使用宿主系统的根 CA 集合。
    RootCAs *x509.CertPool

    // NextProtos 按偏好顺序列出支持的应用层协议。
    // 如果双方都支持 ALPN，选定协议将来自此列表，
    // 如果没有双方都支持的协议，连接会失败。
    // 如果 NextProtos 为空，或对端不支持 ALPN，
    // 则连接会成功，且
    // ConnectionState.NegotiatedProtocol 为空。
    NextProtos []string

    // 除非设置了 InsecureSkipVerify，否则 ServerName 用于验证
    // 返回证书中的主机名。它也会包含在
    // 客户端握手中，以支持虚拟主机，除非其值
    // 是 IP 地址。
    ServerName string

    // ClientAuth 决定服务器的 TLS 客户端
    // 身份认证策略。默认为 NoClientCert。
    ClientAuth ClientAuthType

    // ClientCAs 定义根证书颁发机构集合，
    // 当 ClientAuth 策略要求验证客户端证书时，
    // 服务器使用此集合。
    ClientCAs *x509.CertPool

    // InsecureSkipVerify 控制客户端是否验证服务器的
    // 证书链和主机名。如果 InsecureSkipVerify 为 true，crypto/tls
    // 接受服务器提供的任何证书以及该证书中的任何主机名。
    // 在此模式下，除非使用自定义验证，否则 TLS 容易遭受
    // 中间人攻击。此设置只应供测试使用，
    // 或与 VerifyConnection、VerifyPeerCertificate 配合使用。
    InsecureSkipVerify bool

    // CipherSuites 列出已启用的 TLS 1.0–1.2 密码套件，列表顺序
    // 会被忽略。注意，TLS 1.3 密码套件不可配置。
    //
    // 如果 CipherSuites 为 nil，则使用安全的默认列表。默认密码
    // 套件可能随时间变化。
    CipherSuites []uint16

    // PreferServerCipherSuites 是历史遗留字段，没有作用。
    //
    // 它曾用于控制服务器遵循客户端还是服务器自己的
    // 偏好。现在，服务器会综合推断出的客户端硬件、
    // 服务器硬件及安全性，
    // 选择双方支持的最合适的密码套件。
    //
    // 已弃用：PreferServerCipherSuites 被忽略。
    PreferServerCipherSuites bool // Go 1.1

    // 将 SessionTicketsDisabled 设为 true，可以禁用会话票据和
    // PSK（恢复）支持。注意，在客户端上，如果 ClientSessionCache
    // 为 nil，也会禁用会话票据支持。
    SessionTicketsDisabled bool // Go 1.1

    // TLS 服务器使用 SessionTicketKey 提供会话恢复。
    // 参见 RFC 5077 及 RFC 8446 的 PSK 模式。如果为零值，会在首次
    // 服务器握手之前用随机数据填充。
    //
    // 已弃用：如果此字段保留为零值，会话票据密钥会
    // 每天自动轮换，并在七天后丢弃。若要
    // 自定义轮换计划，或同步为同一主机终止连接的
    // 多台服务器，请使用 SetSessionTicketKeys。
    SessionTicketKey [32]byte // Go 1.1

    // ClientSessionCache 是用于 TLS 会话恢复的 ClientSessionState
    // 条目缓存。只有客户端会使用它。
    ClientSessionCache ClientSessionCache // Go 1.3

    // UnwrapSession 在服务器端被调用，将之前由 [WrapSession]
    // 生成的票据或标识转换为可用会话。
    //
    // UnwrapSession 通常会解密票据中的会话状态
    // （例如通过 [Config.EncryptTicket]），或者将票据作为句柄，
    // 恢复先前存储的状态。它必须使用 [ParseSessionState]
    // 反序列化会话状态。
    //
    // 如果 UnwrapSession 返回错误，则终止连接。如果
    // 返回 (nil, nil)，则忽略该会话。crypto/tls 仍可能
    // 选择不恢复返回的会话。
    UnwrapSession func(identity []byte, cs ConnectionState) (*SessionState, error) // Go 1.21

    // WrapSession 在服务器端被调用，生成会话票据或标识。
    //
    // WrapSession 必须使用 [SessionState.Bytes] 序列化会话状态。
    // 然后可以加密序列化后的状态（例如使用
    // [Config.DecryptTicket]）并将其作为票据，或存储该状态，
    // 返回对应句柄。
    //
    // 如果 WrapSession 返回错误，则终止连接。
    //
    // 警告：返回值会以明文形式暴露在网络上，并提供给客户端。
    // 应用程序负责对其加密和认证（以及轮换密钥），
    // 或返回高熵标识符。如果未能正确完成这些工作，
    // 根据协议版本的不同，当前、过去和未来的连接
    // 都可能受到威胁。
    WrapSession func(ConnectionState, *SessionState) ([]byte, error) // Go 1.21

    // MinVersion 指定可接受的最低 TLS 版本。
    //
    // 当前默认最低版本为 TLS 1.2。本包支持的
    // 最低版本为 TLS 1.0。
    MinVersion uint16 // Go 1.2

    // MaxVersion 指定可接受的最高 TLS 版本。
    //
    // 默认使用本包支持的最高版本，
    // 当前为 TLS 1.3。
    MaxVersion uint16 // Go 1.2

    // CurvePreferences 包含一组支持的密钥交换机制。
    // 字段名因历史原因仍称为椭圆曲线，参见 [CurveID]。
    // 列表顺序被忽略，密钥交换机制会按照内部偏好顺序
    // 从列表中选择。如果为空，则使用
    // 默认值。
    //
    // 自 Go 1.24 起，默认包含 [X25519MLKEM768] 混合
    // 后量子密钥交换。若要禁用它，请显式设置 CurvePreferences，
    // 或使用 GODEBUG=tlsmlkem=0 环境变量。
    //
    // 自 Go 1.26 起，默认还包含 [SecP256r1MLKEM768] 和
    // [SecP384r1MLKEM1024] 混合后量子密钥交换。若要禁用
    // 它们，请显式设置 CurvePreferences，或使用
    // GODEBUG=tlsmlkem=0 或 GODEBUG=tlssecpmlkem=0 环境变量。
    CurvePreferences []CurveID // Go 1.3

    // DynamicRecordSizingDisabled 禁用 TLS 记录大小的自适应调整。
    // 为 true 时，始终使用允许的最大 TLS 记录大小。为
    // false 时，TLS 记录大小可能被调整，
    // 以改善延迟。
    DynamicRecordSizingDisabled bool // Go 1.7

    // Renegotiation 控制支持的重新协商类型。
    // 默认不支持重新协商，这适用于绝大多数应用程序。
    Renegotiation RenegotiationSupport // Go 1.7

    // KeyLogWriter 可选地指定 TLS 主秘密的输出目标，
    // 使用 NSS 密钥日志格式，使 Wireshark 等外部程序
    // 能够解密 TLS 连接。
    // 参见 https://datatracker.ietf.org/doc/draft-ietf-tls-keylogfile/。
    // 使用 KeyLogWriter 会损害安全性，
    // 因此只能用于调试。
    KeyLogWriter io.Writer // Go 1.8

    // EncryptedClientHelloConfigList 是序列化的 ECHConfigList。
    // 如果提供此字段，客户端会使用其中一个 ECHConfig，
    // 尝试通过加密客户端 Hello（ECH）连接服务器。
    //
    // 服务器不使用此字段。若要为服务器配置 ECH，请参阅
    // EncryptedClientHelloKeys 字段。
    //
    // 如果列表不包含有效的 ECH 配置，握手会失败，
    // 并返回错误。
    //
    // 如果设置了 EncryptedClientHelloConfigList，那么 MinVersion（若设置）
    // 必须为 VersionTLS13。
    //
    // 设置 EncryptedClientHelloConfigList 后，只有成功协商
    // ECH，握手才会成功。如果服务器拒绝 ECH，
    // 就返回 ECHRejectionError，其中可能包含服务器建议
    // 使用的新 ECHConfigList。
    //
    // 如果最终的 Encrypted Client Hello RFC 描述的编码发生变化，
    // 此字段的解析方式可能在未来 Go 版本中改变。
    EncryptedClientHelloConfigList []byte // Go 1.23

    // EncryptedClientHelloRejectionVerify 非 nil 时，会在远端服务器
    // 拒绝 ECH 时被调用，用于验证外层 ClientHello 对应的
    // ECH 提供方证书。如果它返回非 nil 错误，
    // 则中止握手并返回该错误。
    //
    // 服务器端不使用此字段。
    //
    // 与 VerifyPeerCertificate 和 VerifyConnection 不同，
    // 调用 EncryptedClientHelloRejectionVerify 之前
    // 不会执行正常证书验证。
    //
    // 如果 EncryptedClientHelloRejectionVerify 为 nil 且 ECH 被拒绝，
    // 则使用 RootCAs 中的根证书验证 ECH 提供方的
    // 公开证书。ECH 被拒绝时，即使设置了 VerifyPeerCertificate 和
    // VerifyConnection，也不会调用它们，而且 InsecureSkipVerify 会被忽略。
    EncryptedClientHelloRejectionVerify func(ConnectionState) error // Go 1.23

    // GetEncryptedClientHelloKeys 非 nil 时，服务器会在
    // 客户端尝试使用 ECH 时调用它。
    //
    // 如果 GetEncryptedClientHelloKeys 非 nil，就会忽略
    // [EncryptedClientHelloKeys]。
    //
    // 如果 GetEncryptedClientHelloKeys 返回错误，则中止握手，
    // 并返回该错误。否则，
    // GetEncryptedClientHelloKeys 必须返回非 nil 的
    // [EncryptedClientHelloKey] 切片，表示可接受的 ECH 密钥。
    //
    // 更多细节参见 [EncryptedClientHelloKeys]。
    GetEncryptedClientHelloKeys func(*ClientHelloInfo) ([]EncryptedClientHelloKey, error) // Go 1.25

    // EncryptedClientHelloKeys 是客户端尝试 ECH 时
    // 使用的 ECH 密钥。
    //
    // 如果设置了 EncryptedClientHelloKeys，那么 MinVersion（若设置）
    // 必须为 VersionTLS13。
    //
    // 如果客户端尝试 ECH，但被服务器拒绝，服务器会
    // 根据 EncryptedClientHelloKeys 中设置了 SendAsRetry 字段
    // 的条目，发送可用于重试的配置列表。
    //
    // 如果 GetEncryptedClientHelloKeys 非 nil，就会忽略
    // EncryptedClientHelloKeys。
    //
    // 在客户端，此字段被忽略。若要为客户端配置 ECH，
    // 请参阅 EncryptedClientHelloConfigList 字段。
    EncryptedClientHelloKeys []EncryptedClientHelloKey // Go 1.24
    // 包含已过滤或未导出的字段
}

type ConnectionState
				
				
			

			
ConnectionState 记录连接的基本 TLS 信息。

			
type ConnectionState struct {
    // Version 是此连接使用的 TLS 版本，例如 VersionTLS12。
    Version uint16 // Go 1.3

    // 握手完成时，HandshakeComplete 为 true。
    HandshakeComplete bool

    // 如果此连接通过会话票据或类似机制从先前会话
    // 成功恢复，DidResume 就为 true。
    DidResume bool // Go 1.1

    // CipherSuite 是为此连接协商的密码套件，例如
    // TLS_ECDHE_ECDSA_WITH_AES_128_GCM_SHA256、TLS_AES_128_GCM_SHA256。
    CipherSuite uint16

    // CurveID 是此连接使用的密钥交换机制。字段名
    // 因历史原因仍称为椭圆曲线，参见 [CurveID]。如果使用传统
    // RSA 密钥交换，此值为零。
    CurveID CurveID // Go 1.25

    // NegotiatedProtocol 是通过 ALPN 协商的应用协议。
    NegotiatedProtocol string

    // NegotiatedProtocolIsMutual 曾用于表示双方都进行了 NPN 协商。
    //
    // 已弃用：此值始终为 true。
    NegotiatedProtocolIsMutual bool

    // ServerName 是客户端发送的服务器名称指示扩展的值。
    // 在服务器端和客户端都可用。
    ServerName string

    // PeerCertificates 是对端发送的已解析证书，
    // 按发送顺序排列。第一个元素是用于验证
    // 此连接的叶子证书。
    //
    // 在客户端，该列表不能为空。在服务器端，如果
    // Config.ClientAuth 既不是 RequireAnyClientCert，也不是
    // RequireAndVerifyClientCert，则可以为空。
    //
    // 不应修改 PeerCertificates 及其中的内容。
    PeerCertificates []*x509.Certificate

    // VerifiedChains 是包含一个或多个证书链的列表，每条链的首个元素
    // 为 PeerCertificates[0]，最后一个元素来自 Config.RootCAs（客户端）
    // 或 Config.ClientCAs（服务器端）。
    //
    // 客户端在 Config.InsecureSkipVerify 为 false 时设置它。
    // 服务器端在 Config.ClientAuth 为 VerifyClientCertIfGiven
    // （且对端提供了证书）或 RequireAndVerifyClientCert 时设置它。
    //
    // 不应修改 VerifiedChains 及其中的内容。
    VerifiedChains [][]*x509.Certificate

    // SignedCertificateTimestamps 是对端通过 TLS 握手
    // 为叶子证书提供的 SCT 列表（如果有）。
    SignedCertificateTimestamps [][]byte // Go 1.5

    // OCSPResponse 是对端为叶子证书提供的
    // 装订在线证书状态协议（OCSP）响应（如果有）。
    OCSPResponse []byte // Go 1.5

    // TLSUnique 包含 "tls-unique" 通道绑定值（参见 RFC 5929
    // 第 3 节）。对于 TLS 1.3 连接，以及不支持扩展主秘密
    // （RFC 7627）的恢复连接，此值为 nil。
    TLSUnique []byte // Go 1.4

    // ECHAccepted 表示客户端是否提出使用加密客户端 Hello，
    // 且服务器接受了它。目前仅在客户端
    // 支持 ECH。
    ECHAccepted bool // Go 1.23

    // HelloRetryRequest 表示：作为服务器时是否发送了 HelloRetryRequest，
    // 或作为客户端时是否收到了 HelloRetryRequest。
    HelloRetryRequest bool // Go 1.26

    // LocalCertificate 是握手期间向对端提供的证书链（如果有）。
    // 只有未通过恢复建立的连接（DidResume 为 false）
    // 才会填充此字段。
    LocalCertificate [][]byte // Go 1.27
    // 包含已过滤或未导出的字段
}

type RenegotiationSupport
				
				1.7
			

			
RenegotiationSupport 枚举对 TLS 重新协商的不同支持级别。TLS 重新协商是指在连接的首次握手之后再次握手。这会显著增加状态机的复杂性，并曾引发大量隐蔽的安全问题。本包不支持主动发起重新协商，但可以启用接受重新协商请求的支持。

即使启用，服务器也不能在不同握手之间改变身份（即叶子证书必须相同）。此外，不允许握手与应用数据传输并发进行，因此重新协商只能用于能够与之同步的协议，例如 HTTPS。

TLS 1.3 未定义重新协商。

const (
    // RenegotiateNever 禁用重新协商。
    RenegotiateNever RenegotiationSupport = iota

    // RenegotiateOnceAsClient 允许远端服务器在每个连接上
    // 请求一次重新协商。
    RenegotiateOnceAsClient

    // RenegotiateFreelyAsClient 允许远端服务器反复
    // 请求重新协商。
    RenegotiateFreelyAsClient

type SessionState struct {

    // crypto/tls 忽略 Extra，但 [SessionState.Bytes] 会编码它，
    // [ParseSessionState] 会解析它。

    // 应用程序只能追加而不能替换，并且必须使用

    // 如果为 true，应用程序可以将其设为 false，
    // 以便即使支持 0-RTT 也不提出使用。

    // 包含已过滤或未导出的字段
